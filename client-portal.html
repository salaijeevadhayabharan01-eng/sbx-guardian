SDK & Code Examples
Copy-paste code for every major platform. Replace YOUR_WEBHOOK_URL with your actual URL.
Python (Universal)
import requests, time WEBHOOK_URL = "YOUR_WEBHOOK_URL" SYSTEM_ID = "robot-arm-01" def log_event(event_type, severity, message, data={}): requests.post(WEBHOOK_URL, json={ "system_id": SYSTEM_ID, "type": event_type, "severity": severity, "message": message, "health": 95, "load": 42.5, "data": data }) # Log a heartbeat every 30 seconds while True: log_event("heartbeat", "info", "System operational") time.sleep(30) # Log an anomaly log_event("anomaly", "warning", "Temperature spike detected", { "temperature": 87.3, "threshold": 80.0 })
Node.js / JavaScript
const WEBHOOK_URL = 'YOUR_WEBHOOK_URL'; async function logEvent(type, severity, message, data = {}) { await fetch(WEBHOOK_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ system_id: 'robot-001', type, severity, message, health: 95, load: 42.5, data }) }); } // Heartbeat setInterval(() => logEvent('heartbeat', 'info', 'Operational'), 30000); // On error logEvent('error', 'critical', 'E-Stop triggered', { cause: 'obstruction' });
ROS2 (Python Node)
import rclpy, requests from rclpy.node import Node class SBXLogger(Node): WEBHOOK = "YOUR_WEBHOOK_URL" def __init__(self): super().__init__('sbx_logger') self.create_timer(30.0, self.heartbeat) def heartbeat(self): requests.post(self.WEBHOOK, json={ "system_id": "ros2-robot-01", "type": "heartbeat", "severity": "info", "message": "ROS2 node operational" }) def log_decision(self, decision, confidence): requests.post(self.WEBHOOK, json={ "system_id": "ros2-robot-01", "type": "ai_decision", "severity": "info", "message": f"Decision: {decision}", "data": {"confidence": confidence} }) def main(): rclpy.init() rclpy.spin(SBXLogger()) if __name__ == '__main__': main()
Arduino / Embedded C
#include <WiFi.h> #include <HTTPClient.h> const char* WEBHOOK = "YOUR_WEBHOOK_URL"; void logEvent(const char* type, const char* msg) { if(WiFi.status() != WL_CONNECTED) return; HTTPClient http; http.begin(WEBHOOK); http.addHeader("Content-Type", "application/json"); String body = "{\"system_id\":\"arduino-01\",\"type\":\""; body += type; body += "\",\"severity\":\"info\",\"message\":\""; body += msg; body += "\"}"; http.POST(body); http.end(); } void loop() { logEvent("heartbeat", "Operational"); delay(30000); }
